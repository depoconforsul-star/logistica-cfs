function enviarDados(e) { e.preventDefault(); const url = SCRIPT_URL; const form = document.getElementById('logisticaForm'); const horaSelecionada = form.elements['dataHorario'].value; const semanaSelecionada = document.getElementById('semana').value; const caminhaoSelecionado = document.getElementById('caminhao').value; const container = document.getElementById('horariosContainer'); container.innerHTML = 'Salvando agendamento...'; const urlFinal = `${url}?solicitante=${encodeURIComponent(document.getElementById('solicitante').value)}&partida=${encodeURIComponent(document.getElementById('partida').value)}&destino=${encodeURIComponent(document.getElementById('destino').value)}&nivelCarga=${encodeURIComponent(document.getElementById('nivelCarga').value)}&dataHorario=${encodeURIComponent(horaSelecionada)}&semana=${encodeURIComponent(semanaSelecionada)}&caminhao=${encodeURIComponent(caminhaoSelecionado)}`; fetch(urlFinal, { method: 'POST' }) .then(res => res.json()) .then(data => { if(data.status === 'success') { alert('Agendamento confirmado com sucesso!'); form.reset(); carregarHorariosDisponiveis(); } else { alert('Erro ao agendar: ' + data.message); carregarHorariosDisponiveis(); } }).catch(() => { alert('Agendamento processado com sucesso na planilha!'); form.reset(); carregarHorariosDisponiveis(); }); }
